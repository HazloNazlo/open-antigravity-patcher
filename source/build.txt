Windows (x64):
pyinstaller --onefile --uac-admin --icon=icon.ico --name="Open_AG_Patcher_Windows_x64" --noupx --clean --version-file=version.txt --hidden-import=packaging --hidden-import=packaging.version --hidden-import=packaging.specifiers --hidden-import=packaging.requirements main.py

Windows (arm64): та же команда, --name="Open_AG_Patcher_Windows_arm64"

Linux (x64):
pyinstaller --onefile --icon=icon.ico --name="Open_AG_Patcher_Linux_x64" --hidden-import=packaging --hidden-import=packaging.version --hidden-import=packaging.specifiers --hidden-import=packaging.requirements --clean main.py

Linux (arm64): та же команда, --name="Open_AG_Patcher_Linux_arm64"

macOS universal2:
pyinstaller --onefile --name="Open_AG_Patcher_macOS" --target-arch universal2 --hidden-import=packaging --hidden-import=packaging.version --hidden-import=packaging.specifiers --hidden-import=packaging.requirements --clean main.py
codesign --force --sign - --timestamp=none dist/Open_AG_Patcher_macOS
codesign --verify --strict dist/Open_AG_Patcher_macOS

macOS (только arm64, если нет universal2-Python):
pyinstaller --onefile --name="Open_AG_Patcher_macOS" --target-arch arm64 --hidden-import=packaging --hidden-import=packaging.version --hidden-import=packaging.specifiers --hidden-import=packaging.requirements --clean main.py
